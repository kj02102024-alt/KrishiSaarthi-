        function poolHarvestIntoBatch(batchId) {
            const batch = logisticsBatches.find(b => b.id === batchId);
            if(batch) {
                if(batch.filled >= batch.capacity) {
                    showToast("Batch capacity already full!", "error");
                    return;
                }
                batch.filled = Math.min(batch.capacity, batch.filled + 10);
                const tag = `${currentUser.name} (10 Qtl)`;
                if(!batch.farmers.includes(tag)) batch.farmers.push(tag);
                localStorage.setItem('ks_logistics_batches_v2', JSON.stringify(logisticsBatches));
                renderLogisticsList();
                showToast(`Added 10 Quintal into ${batchId}! Space locked.`, 'success');
            }
        }

        // error.code is what actually tells you which setting to fix — log it,
        // don't just log the message (which is often generic).

        showToast(`OTP failed: ${error.code || error.message}`, "error");
    }
